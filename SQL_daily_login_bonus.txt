CREATE TABLE IF NOT EXISTS `site_daily_login_claims` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `player_id` int(11) NOT NULL,
  `week_key` char(10) NOT NULL,
  `day_number` tinyint(3) unsigned NOT NULL,
  `claim_date` date NOT NULL,
  `claimed_at` datetime NOT NULL,
  `rewards_json` text NOT NULL,
  PRIMARY KEY (`id`),
  UNIQUE KEY `uniq_daily_login_player_week_day` (`player_id`, `week_key`, `day_number`),
  UNIQUE KEY `uniq_daily_login_player_claim_date` (`player_id`, `claim_date`),
  KEY `idx_daily_login_player_week` (`player_id`, `week_key`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;
