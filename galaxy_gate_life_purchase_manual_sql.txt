-- Manual SQL for Galaxy Gate extra life purchase pricing.
-- Adds a per-player/per-gate counter used to double the next extra life price.
-- Execute manually on the Andromeda database. Do not add this to andromeda.sql.

SET @column_exists := (
    SELECT COUNT(*)
    FROM INFORMATION_SCHEMA.COLUMNS
    WHERE TABLE_SCHEMA = DATABASE()
      AND TABLE_NAME = 'player_galaxy_gates'
      AND COLUMN_NAME = 'life_purchases'
);

SET @sql := IF(
    @column_exists = 0,
    'ALTER TABLE `player_galaxy_gates` ADD COLUMN `life_purchases` INT NOT NULL DEFAULT 0 AFTER `lives`',
    'SELECT ''player_galaxy_gates.life_purchases already exists'' AS message'
);

PREPARE stmt FROM @sql;
EXECUTE stmt;
DEALLOCATE PREPARE stmt;

UPDATE `player_galaxy_gates`
SET `life_purchases` = 0
WHERE `on_map` = 0;
