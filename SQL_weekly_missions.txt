CREATE TABLE IF NOT EXISTS site_weekly_missions (
    id INT AUTO_INCREMENT PRIMARY KEY,
    code VARCHAR(64) NOT NULL UNIQUE,
    title VARCHAR(180) NOT NULL,
    description TEXT NOT NULL,
    rotation_group CHAR(1) NOT NULL,
    slot TINYINT NOT NULL,
    difficulty VARCHAR(32) NOT NULL DEFAULT 'Hard',
    recommended_level INT NOT NULL DEFAULT 12,
    reward_uridium BIGINT NOT NULL DEFAULT 0,
    reward_experience BIGINT NOT NULL DEFAULT 0,
    reward_honor BIGINT NOT NULL DEFAULT 0,
    reward_ucb100 BIGINT NOT NULL DEFAULT 0,
    reward_rsb75 BIGINT NOT NULL DEFAULT 0,
    reward_seprom BIGINT NOT NULL DEFAULT 0,
    enabled TINYINT(1) NOT NULL DEFAULT 1,
    updated_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    KEY idx_weekly_rotation (enabled, rotation_group, slot)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

CREATE TABLE IF NOT EXISTS site_weekly_mission_objectives (
    id INT AUTO_INCREMENT PRIMARY KEY,
    mission_id INT NOT NULL,
    objective_type VARCHAR(32) NOT NULL,
    target_key VARCHAR(96) NOT NULL,
    objective_label VARCHAR(220) NOT NULL,
    required_amount INT NOT NULL DEFAULT 1,
    sort_order INT NOT NULL DEFAULT 0,
    KEY idx_weekly_mission_sort (mission_id, sort_order),
    KEY idx_weekly_objective (objective_type, target_key)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

CREATE TABLE IF NOT EXISTS site_player_weekly_missions (
    id BIGINT AUTO_INCREMENT PRIMARY KEY,
    player_id INT NOT NULL,
    mission_id INT NOT NULL,
    week_key VARCHAR(10) NOT NULL,
    status VARCHAR(24) NOT NULL DEFAULT 'in_progress',
    accepted_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    claimed_at DATETIME DEFAULT NULL,
    UNIQUE KEY uniq_player_weekly_mission (player_id, mission_id, week_key),
    KEY idx_player_week_status (player_id, week_key, status),
    KEY idx_weekly_mission_week (mission_id, week_key)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

CREATE TABLE IF NOT EXISTS site_player_weekly_mission_progress (
    player_id INT NOT NULL,
    mission_id INT NOT NULL,
    week_key VARCHAR(10) NOT NULL,
    objective_type VARCHAR(32) NOT NULL,
    target_key VARCHAR(96) NOT NULL,
    current_amount INT NOT NULL DEFAULT 0,
    updated_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    PRIMARY KEY (player_id, mission_id, week_key, objective_type, target_key),
    KEY idx_weekly_progress_player (player_id, week_key),
    KEY idx_weekly_progress_objective (objective_type, target_key)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;
