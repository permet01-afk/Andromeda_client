-- Andromeda Pilot Bio / Skill Tree V1.3 manual SQL.
-- Make a full database backup before running this file.
-- Run and validate locally/test first, then only run live after validation.
-- This file creates/seeds Pilot Bio tables and migrates existing players to Pilot Bio.
-- This file is not imported automatically and does not modify andromeda.sql.
-- It does not modify users.skilltree, users.hp_lvl, or users.max_hp.

-- STEP 1 - Schema and seed.

CREATE TABLE IF NOT EXISTS pilot_bio_nodes (
  id INT NOT NULL AUTO_INCREMENT,
  node_code VARCHAR(64) NOT NULL,
  display_name VARCHAR(96) NOT NULL,
  description_en VARCHAR(255) NOT NULL,
  track ENUM('defense','progression','offense') NOT NULL,
  visual_slot TINYINT UNSIGNED NOT NULL,
  max_level TINYINT UNSIGNED NOT NULL DEFAULT 1,
  status ENUM('active','later','visual_only','disabled') NOT NULL DEFAULT 'later',
  effect_key VARCHAR(64) DEFAULT NULL,
  effect_values_json TEXT DEFAULT NULL,
  prerequisites_json TEXT DEFAULT NULL,
  created_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
  updated_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  PRIMARY KEY (id),
  UNIQUE KEY uq_pilot_bio_nodes_code (node_code),
  UNIQUE KEY uq_pilot_bio_nodes_slot (visual_slot),
  KEY idx_pilot_bio_nodes_track_slot (track, visual_slot)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

CREATE TABLE IF NOT EXISTS player_pilot_bio_state (
  user_id INT NOT NULL,
  research_points INT UNSIGNED NOT NULL DEFAULT 0,
  spent_points INT UNSIGNED NOT NULL DEFAULT 0,
  reset_count INT UNSIGNED NOT NULL DEFAULT 0,
  migrated_from_legacy TINYINT(1) NOT NULL DEFAULT 0,
  legacy_skilltree_snapshot VARCHAR(255) DEFAULT NULL,
  legacy_hp_lvl_snapshot INT DEFAULT NULL,
  legacy_shreg_refund_points INT UNSIGNED NOT NULL DEFAULT 0,
  created_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
  updated_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  PRIMARY KEY (user_id),
  CONSTRAINT fk_player_pilot_bio_state_user
    FOREIGN KEY (user_id) REFERENCES users(id)
    ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

CREATE TABLE IF NOT EXISTS player_pilot_bio_levels (
  user_id INT NOT NULL,
  node_code VARCHAR(64) NOT NULL,
  level TINYINT UNSIGNED NOT NULL DEFAULT 0,
  created_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
  updated_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  PRIMARY KEY (user_id, node_code),
  KEY idx_player_pilot_bio_levels_node (node_code),
  CONSTRAINT fk_player_pilot_bio_levels_user
    FOREIGN KEY (user_id) REFERENCES users(id)
    ON DELETE CASCADE,
  CONSTRAINT fk_player_pilot_bio_levels_node
    FOREIGN KEY (node_code) REFERENCES pilot_bio_nodes(node_code)
    ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

CREATE TABLE IF NOT EXISTS pilot_bio_point_costs (
  point_number INT UNSIGNED NOT NULL,
  logfile_cost INT UNSIGNED NOT NULL,
  PRIMARY KEY (point_number)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- STEP 2 - Automatic global migration.
-- This migrates players that do not already have a Pilot Bio state.
-- hp_lvl mapping used by V1:
--   hp_lvl 1  (+5,000 HP)  -> Ship Hull I level 1
--   hp_lvl 2  (+10,000 HP) -> Ship Hull I level 2
--   hp_lvl 3 or 4          -> Ship Hull I level 2 + Ship Hull II level 1
--   hp_lvl 5 to 9          -> Ship Hull I level 2 + Ship Hull II level 2
--   hp_lvl 10 (+50,000 HP) -> Ship Hull I level 2 + Ship Hull II level 3
-- Ship Hull I and II are additive in V1.3, so max Pilot Bio Ship Hull gives +60,000 HP total.
-- A legacy hp_lvl 10 pilot gains +10,000 HP compared to old full Ship Integrity after migration.
-- hp_lvl 4 and 6-9 are rounded down to the lower Pilot Bio HP tier.
-- Optional compensation for those intermediate HP tiers can be decided later.
-- Shield Regeneration is not active in V1, so old shreg levels are refunded as Research Points.
-- Rocket bonuses and legacy skilltree hp are ignored in V1.

START TRANSACTION;

INSERT INTO player_pilot_bio_levels (user_id, node_code, level)
SELECT u.id, 'ship_hull_i',
  CASE
    WHEN u.hp_lvl >= 2 THEN 2
    WHEN u.hp_lvl >= 1 THEN 1
    ELSE 0
  END
FROM users u
WHERE u.hp_lvl > 0
  AND NOT EXISTS (SELECT 1 FROM player_pilot_bio_state s WHERE s.user_id = u.id)
ON DUPLICATE KEY UPDATE level = GREATEST(player_pilot_bio_levels.level, VALUES(level));

INSERT INTO player_pilot_bio_levels (user_id, node_code, level)

  AND NOT EXISTS (SELECT 1 FROM player_pilot_bio_state s WHERE s.user_id = u.id)

WHERE NOT EXISTS (SELECT 1 FROM player_pilot_bio_state s WHERE s.user_id = u.id);

SET spent_points = (
  SELECT COALESCE(SUM(level), 0)
  FROM player_pilot_bio_levels l
  WHERE l.user_id = s.user_id

WHERE s.migrated_from_legacy = 1;

-- STEP 3 - Verification queries.

SELECT COUNT(*) AS migrated_players
FROM player_pilot_bio_state;

SELECT COUNT(*) AS non_migrated_players
FROM users u
LEFT JOIN player_pilot_bio_state s ON s.user_id = u.id
WHERE s.user_id IS NULL;

SELECT n.node_code, n.display_name, n.status, COUNT(l.user_id) AS pilots_with_level, COALESCE(SUM(l.level), 0) AS total_levels
FROM pilot_bio_nodes n
LEFT JOIN player_pilot_bio_levels l ON l.node_code = n.node_code AND l.level > 0
GROUP BY n.node_code, n.display_name, n.status
ORDER BY n.visual_slot ASC;

SELECT COALESCE(SUM(spent_points), 0) AS total_spent_points, COALESCE(SUM(research_points), 0) AS total_research_points
FROM player_pilot_bio_state;

SELECT user_id, research_points, spent_points, migrated_from_legacy, legacy_hp_lvl_snapshot, legacy_skilltree_snapshot, legacy_shreg_refund_points
FROM player_pilot_bio_state
ORDER BY user_id ASC
LIMIT 25;

SELECT s.user_id, s.research_points, s.spent_points, l.node_code, l.level
FROM player_pilot_bio_state s
LEFT JOIN player_pilot_bio_levels l ON l.user_id = s.user_id
WHERE s.user_id = 3
ORDER BY l.node_code ASC;
