-- Andromeda Pilot Bio / Skill Tree manual SQL proposal
-- Proposal only. Do not run before the Pilot Bio patch is validated.
-- This file intentionally does not modify andromeda.sql.

CREATE TABLE IF NOT EXISTS pilot_bio_nodes (
  id INT UNSIGNED NOT NULL AUTO_INCREMENT,
  node_code VARCHAR(64) NOT NULL,
  display_name VARCHAR(96) NOT NULL,
  track ENUM('defense','progression','offense','custom') NOT NULL,
  sort_order INT UNSIGNED NOT NULL DEFAULT 0,
  max_level TINYINT UNSIGNED NOT NULL DEFAULT 1,
  status ENUM('active','disabled','later','visual_only','legacy_removed') NOT NULL DEFAULT 'later',
  effect_key VARCHAR(64) DEFAULT NULL,
  effect_values_json TEXT DEFAULT NULL,
  prerequisites_json TEXT DEFAULT NULL,
  description_en VARCHAR(255) NOT NULL,
  created_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
  updated_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  PRIMARY KEY (id),
  UNIQUE KEY uq_pilot_bio_nodes_code (node_code),
  KEY idx_pilot_bio_nodes_track_sort (track, sort_order)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

CREATE TABLE IF NOT EXISTS player_pilot_bio_state (
  user_id INT UNSIGNED NOT NULL,
  research_points INT UNSIGNED NOT NULL DEFAULT 0,
  spent_points INT UNSIGNED NOT NULL DEFAULT 0,
  reset_count INT UNSIGNED NOT NULL DEFAULT 0,
  migrated_from_legacy TINYINT(1) NOT NULL DEFAULT 0,
  legacy_skilltree_snapshot VARCHAR(255) DEFAULT NULL,
  legacy_hp_lvl_snapshot INT DEFAULT NULL,
  created_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
  updated_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  PRIMARY KEY (user_id),
  CONSTRAINT fk_player_pilot_bio_state_user
    FOREIGN KEY (user_id) REFERENCES users(id)
    ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

CREATE TABLE IF NOT EXISTS player_pilot_bio_levels (
  user_id INT UNSIGNED NOT NULL,
  node_code VARCHAR(64) NOT NULL,
  level TINYINT UNSIGNED NOT NULL DEFAULT 0,
  created_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
  updated_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  PRIMARY KEY (user_id, node_code),
  KEY idx_player_pilot_bio_levels_node (node_code),
  CONSTRAINT fk_player_pilot_bio_levels_user
    FOREIGN KEY (user_id) REFERENCES users(id)
    ON DELETE CASCADE,
  CONSTRAINT fk_player_pilot_bio_levels_node
    FOREIGN KEY (node_code) REFERENCES pilot_bio_nodes(node_code)
    ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

CREATE TABLE IF NOT EXISTS pilot_bio_point_costs (
  point_number INT UNSIGNED NOT NULL,
  logfile_cost INT UNSIGNED NOT NULL,
  PRIMARY KEY (point_number)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;
