CREATE TABLE IF NOT EXISTS player_titles (
  player_id INT NOT NULL,
  title_key VARCHAR(32) NOT NULL,
  title_scope ENUM('permanent', 'temporary') NOT NULL DEFAULT 'permanent',
  source VARCHAR(64) NOT NULL DEFAULT '',
  unlocked_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  expires_at DATETIME DEFAULT NULL,
  revoked_at DATETIME DEFAULT NULL,
  PRIMARY KEY (player_id, title_key),
  KEY idx_player_titles_active (player_id, revoked_at, expires_at),
  KEY idx_player_titles_key (title_key)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

CREATE TABLE IF NOT EXISTS player_title_selection (
  player_id INT NOT NULL,
  selected_title_key VARCHAR(32) NOT NULL DEFAULT '',
  updated_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  PRIMARY KEY (player_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

CREATE TABLE IF NOT EXISTS player_title_progress (
  player_id INT NOT NULL,
  progress_key VARCHAR(64) NOT NULL,
  current_amount INT NOT NULL DEFAULT 0,
  updated_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  PRIMARY KEY (player_id, progress_key),
  KEY idx_title_progress_key (progress_key)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

CREATE TABLE IF NOT EXISTS title_runtime_state (
  state_key VARCHAR(64) NOT NULL,
  title_key VARCHAR(32) NOT NULL,
  holder_type ENUM('none', 'npc', 'player') NOT NULL DEFAULT 'none',
  holder_player_id INT NOT NULL DEFAULT 0,
  holder_npc_id INT NOT NULL DEFAULT 0,
  holder_map_id INT NOT NULL DEFAULT 0,
  assigned_at DATETIME DEFAULT NULL,
  expires_at DATETIME DEFAULT NULL,
  updated_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  PRIMARY KEY (state_key),
  KEY idx_title_runtime_holder_player (holder_player_id),
  KEY idx_title_runtime_holder_npc (holder_npc_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;
